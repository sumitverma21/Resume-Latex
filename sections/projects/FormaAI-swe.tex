\begin{cvproject}{\uline{Forma AI}}{React, Node.js, Express, LangChain, Zustand, MongoDB}
    \item Engineered an AI-driven dynamic form engine with \textbf{React, Zustand, Node.js, and Express}, dynamically rendering UI forms from JSON schemas stored in \textbf{MongoDB}.
    \item Integrated \textbf{LangChain} and LLM extraction pipelines to parse natural language incident descriptions into validated, schema-compliant JSON payloads.
    \item Architected dynamic form state management supporting deep conditional branching logic (\textbf{showIf} rules) and partial state save/resume workflows.
    \item Implemented schema-aware validation gates and anti-hallucination confidence scoring for deterministic data submission.
\end{cvproject}
